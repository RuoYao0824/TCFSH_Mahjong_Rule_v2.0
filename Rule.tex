\section*{第一章 \ 基本精神}
所有選手應基於以下基本精神進行對局
\begin{itemize}
    \item 所有選手須盡力確保比賽之公平
    \item 所有選手皆負有審判之義務
    \item 必須絕對服從裁判之判決
    \item 必須以追求自身之利益為目的進行比賽
    \item 必須盡力使對局得以流暢進行
\end{itemize}

\section*{第二章 \ 競技的基本}
\begin{itemize}
    \item[1] 競技的構成
    \item[] 競技為一桌四人進行
    \item[2] 牌
    \item[] 競技用牌使用一套136張,共2種 各牌的稱呼如下圖
        \begin{figure}[H]
        \centering
        \includegraphics[width=1.0\textwidth]{牌種類.png}
        \end{figure}
    \item[3] 骰子
    \item[] 骰子使用2個六面骰。骰子合計的值稱為「出目」,對應方位如下:
    \begin{itemize}
        \item[$\bullet$]「2, 6, 10」 下家
        \item[$\bullet$]「3, 7, 11」 對家
        \item[$\bullet$]「4, 8, 12」 上家
        \item[$\bullet$]「5, 9」 \hspace{0.5cm} 自家
    \end{itemize}
    \item[4] 點棒
    \begin{itemize}
        \item[(1)] 分數的移動使用點棒
        \item[(2)]一套點棒含10000點1根、5000點1根、1000點4根、500點1根、100點5根共25000點。每位選手初始持有一套點棒, 若點棒數量有誤,選手有義務於對局開始前提出。
    \end{itemize}
    \item[5] 自動牌桌
    \item[] 對局使用具自動升起牌牆功能的麻將桌, 對局開始前應確認自動牌桌之功能是否正常。
    \item[6] 電梯線
    \item[] 牌牆升起的位置稱為電梯線
    \item[7] 裁判
    \item[] 對局中負責裁決的人稱為裁判。雖須絕對服從裁判之判決, 但至隔天為止皆可以書面方式提出異議。
    \item[8] 壁牌
    \item[] 碼好碼齊狀態的牌稱為壁牌
    \item[9] 墩
    \item[] 牌牆上下兩張為一組的單位稱為墩
    \item[10] 開門
    \item[] 每局開始時, 從牌牆中取配牌並開啟缺口的行為稱為開門。開門位置規定如下:
    \begin{itemize}
        \item [(1)] 以出目所指定方位之牌牆為開門對象
        \item[(2)] 自該牌牆面對桌子中央的左側數起, 留下出目數量的墩
    \end{itemize}
    \item[11] 王牌
    \begin{itemize}
        \item[(1)] 從開門位置順時針數起留下的共14張牌稱為王牌
        \item[(2)] 每當進行槓牌時, 將當前作為海底牌的牌補充進王牌中
        \item[(3)] 王牌中自順時針數起的前4張牌稱為嶺上牌
        \item[(4)] 嶺上牌的取得順序為 第一墩上層→第一墩下層→第二墩上層→第二墩下層
    \end{itemize}
    \item[12] 牌河
    \item[] 排列捨牌處稱為牌河
    \item[13] 海底牌
    \item[] 牌正前方最後的那張可摸之牌稱為海底牌
    \item[14] 河底牌
    \item[] 一局中切出的最後一張捨牌稱為河底牌
    \item[15] 手番
    \item[] 從摸牌或副露開始,到切牌為止的期間稱為手番
    \item[16] 牌的組合
    \item[] 牌的組合與稱呼規定如下:
        \begin{figure}[H]
        \includegraphics[width=0.8\textwidth]{牌組合.png}
        \end{figure}
    \item[17] 配牌
    \item[] 親家切出第一打前各家的手牌稱為配牌
    \item[18] 手牌
    \item[] 除自摸牌、嶺上自摸牌、副露牌之外,屬於自己的牌稱為手牌
    \item[19] 倒牌
    \item[] 將手牌全部正面朝上公開展示的行為稱為倒牌
    \item[20] 副露牌
    \item[] 因副露而亮出的面子,以及因加槓成立的明槓子、暗槓子,通稱為副露牌
    \item[21] 門前清
    \item[] 沒有任何副露的狀態稱為門前清
    \item[22] 家
        \begin{itemize}
            \item[(1)] 對局開始時的親家稱為起家
            \item[(2)] 從親家開始順時針分別稱為東家、南家、西家、北家
            \item[(3)] 東家也稱為莊家
            \item[(4)] 東家以外的三人稱為閒家
            \item[(5)] 自己也稱為自家
            \item[(6)] 自己以外的三人稱為他家
            \item[(7)] 左邊座位的人稱為上家、正面座位稱為對家、右邊座位稱為下家
        \end{itemize}
    \item[23] 寶牌
        \begin{itemize}
            \item[(1)] 面對王牌末尾席位的人,要將第1張嶺上牌移到第2張嶺上牌旁
            \item[(2)] 隨後,將王牌末尾數來第3墩上層的牌翻開。這張牌稱為寶牌指示牌
            \item[(3)] 根據寶牌指示牌,寶牌規定如下:
            \begin{itemize}
                \item[$\bullet$] 數牌:同花色依1→2→ \dots →8 →9 →1的順序之下一張
                \item[$\bullet$] 風牌:依東→南→西→北→東的順序之下一張
                \item[$\bullet$] 三元牌:依白→發→中→白的順序之下一張
            \end{itemize}
            \item[(4)] 和牌時,手牌中每有一張寶牌多加算一翻 (包括和了牌)
            \item[(5)] 每次有槓牌時寶牌就會增加。第1次槓牌時,將當前顯示的寶牌指示牌鄰側 (與王牌末尾相反方向)的牌翻開,這張成為新寶牌指示牌。此後以此類推。
            \item[(6)] 所有寶牌指示牌皆由面對該牌席位的人負責翻開
            \item[(7)] 立直後的和了可加算裏寶牌。當前顯示的寶牌指示牌之下層牌即為裏寶牌指示牌
            \item[(8)] 每套競技用牌中皆有赤五萬、赤五筒、赤五索各1張,這些牌稱為赤寶牌。赤寶牌視為寶牌
            \item[(9)] 所有寶牌皆不得因任何理由放棄
        \end{itemize}
    \item[24] 暴露牌
    \item[] 原本不該公開的牌不小心讓他人看見,稱為暴露牌。暴露牌必須向全員公開,並放回原位繼續對局
\end{itemize}
\section*{第三章 \ 競技的進行}
\begin{itemize}
    \item[25] 半莊的準備
    \begin{itemize}
        \item[(1)] 四名對局者任意就座。任意一人將「東·南·西·北」4張牌與筒子「1・2」2張牌,共計6張牌蓋著搓洗打亂,並排成橫向一列
        \item[(2)] 排牌者的對家擲骰子。依點數指定的席位成為「假東」位置,再由該人擲骰子1次
        \item[(3)] 將排好的6張牌翻開,將數牌移動到邊緣
        \begin{figure}[H]
        \includegraphics[width=0.5\textwidth]{抓位.png}
        \end{figure}
        \item[(4)] 自前述第2項第二擲骰所指定的席位之人開始,依順時針方向拿取風牌。若此時點數為奇數,從「1」的那端開始拿;若為偶數,從「2」的那端開始拿
        \item[(5)] 抽到「東」的人坐到「假東」位置。此後,每人各自坐到抽到風牌的對應席位
        \item[(6)] 假東者擲骰子點數指定的席位者為假親,假親再擲骰子。此點數指定的席位者即為「起家」
    \end{itemize}
    \item[26] 半莊的進行
    \begin{itemize}
        \item[(1)] 莊家和了或莊家聽牌流局時,莊家位不變。稱為連莊
        \item[(2)] 閒家和了或莊家沒聽牌流局時,莊家移至下家。稱為輪莊
        \item[(3)] 全員當過1次莊家為東場,當過2次莊家為南場
        \item[(4)] 隨著莊家的移動,局的稱呼依序規定如下:東1局→東2局→東3局→東4局→南1局→南2局→南3局→南4局 
        \item[(5)] 東場與南場合稱為半莊。半莊結束後進行成績結算
        \item[(6)] 半莊的開始與結束規定如下:
        \begin{itemize}
            \item[$\bullet$] 東1局開始即為半莊開始
            \item[$\bullet$] 連莊或輪莊重複進行至南4局
            \item[$\bullet$] 南4局結束且沒有連莊時,半莊即告結束
        \end{itemize}
    \end{itemize}
    \item[27] 局的進行
    \begin{itemize}
        \item[(1)] 配牌整理好後,反覆進行摸打及副露,直到有人和了或王牌剩餘14張為止
        \item[(2)] 摸打由莊家開始依順時針進行。拿牌則依逆時針取得
        \item[(3)] 局的開始與結束規定如下
        \begin{itemize}
            \item[$\bullet$] 莊家按下設定鈕、牌牆升起的行為即為局的開始
            \item[$\bullet$] 原則上依 [東家・西家] → [南家・北家] 的順序推牌牆
            \item[$\bullet$] 由莊家擲骰子1次決定開門位置
            \item[$\bullet$] 從莊家開始依順時針順序拿取配牌
            \item[$\bullet$] 與前述拿牌並行,放下嶺上牌,並翻開寶牌指示牌
            \item[$\bullet$] 莊家打出第一手,之後移至下家摸打。如此循環
            \item[$\bullet$] 和了成立後的點棒交付完畢或流局,即為局的結束
        \end{itemize}
    \end{itemize}
    \item[28] 局間
    \begin{itemize}
        \item[(1)] 下一局的莊家按下設定鈕,全體對局者一起把牌推入孔中
        \item[(2)] 因不可抗力理由需要中途離席者,可在局結束至前項規定的期間內,獲得裁判允許後中途離席
    \end{itemize}
    \item[29] 牌牆的獨立性
    \item[] 除自摸外,不得觸碰面前或他家的牌牆
    \item[30] 審判義務
    \item[] 發生涉及扣分處罰 (包含注意)的情況時,不得繼續進行該局
    \item[31] 可動範圍
    \item[] 非自身手番時,手不得越過桌內的對角線 (下圖紅線位置)
        \begin{figure}[H]
        \includegraphics[width=0.5\textwidth]{可動範圍.png}
        \end{figure}
    \item[32] 發聲行為
    \begin{itemize}
        \item[(1)] 競技進行所必需的發聲,必須讓同桌全員都能明確聽見
        \item[(2)] 必須限制在競技進行所需的最低限度內容
    \end{itemize}
\end{itemize}
\section*{第四章 \ 競技行為}
\begin{itemize}
    \item[33] 取配牌
    \begin{itemize}
        \item[(1)] 取配牌指取得配牌並將其整理為一排的行為
        \item[(2)] 從莊家開始依逆時針順序進行
        \item[(3)] 牌依順時針方向取得
        \item[(4)] 當上家的手收回至電梯線前時,自家方可開始動作
        \item[(5)] 若使用自動配牌功能,以親家的第一自摸視為配牌完成
        \item[(6)] 不使用自動配牌功能的牌桌,依以下步驟進行:
        \begin{itemize}
            \item[$\bullet$] 從開門位置起,每人每次取得2墩,各取得3次
            \item[$\bullet$] 莊家取得第1墩和第3墩的上層牌
            \item[$\bullet$] 閒家各取得1張牌
        \end{itemize}
    \end{itemize}
    \item[34] 自摸
    \begin{itemize}
        \item[(1)] 從牌牆中取得1張牌加入自己手牌的行為稱為自摸
        \item[(2)] 自摸必須在上家打牌 (立直時為供託點棒放置)完成,且確認上家的手收回至電梯線以內後方可開始
        \item[(3)] 手伸出電梯線外即視為自摸動作的開始
        \item[(4)] 一旦開始自摸者,不得中斷或變更該行為
        \item[(5)] 經由自摸取得的牌稱為自摸牌
    \end{itemize}
    \item[35] 打牌
    \begin{itemize}
        \item[(1)] 從手牌或自摸牌中選擇任意1張牌捨棄至牌河的行為稱為打牌
        \item[(2)] 打牌必須捨棄在第37條所規定的位置
        \item[(3)] 打牌時必須迅速放開手,使他家能同時辨識該牌
        \item[(4)] 牌觸及牌河的瞬間即視為打牌完成,不得更改捨牌
    \end{itemize}
    \item[36] 摸打
    \begin{itemize}
        \item[(1)] 從自摸開始到打牌結束的整套行為合稱為摸打
        \item[(2)] 自摸牌在進行打牌行為之前,不得混入非公開的手牌中,且必須明確展現並未混入手中
    \end{itemize}
    \item[37] 捨牌
    \begin{itemize}
        \item[(1)] 因打牌而捨棄的牌稱為捨牌
        \item[(2)] 捨牌依下圖模式排列
        \begin{figure}[H]
            \includegraphics[width=0.5\textwidth]{捨牌.png}
        \end{figure}
        \item[(3)] 捨牌每列為6張,第二列起排在下方,以此類推
        \item[(4)] 不得無故觸碰捨牌
    \end{itemize}
    \item[38] 立直
    \begin{itemize}
        \item[(1)] 立直宣告牌若沒有被他人榮和,立直即宣告成立。步驟規定於第13項
        \item[(2)] 本條第13項第3款之後,不得取消立直
        \item[(3)] 只有門前清狀態可以宣告立直
        \item[(4)] 振聽立直或已沒有自摸機會的立直也可以宣告
        \item[(5)] 唯在河底牌不得宣告立直
        \item[(6)] 若立直宣告牌被他人副露,將下一張捨牌橫放
        \item[(7)] 若前項橫放的捨牌又被副露,再將下一張捨牌橫放,以此類推直到未被副露為止
        \item[(8)] 立直後必須直接切出摸到的自摸牌。但和了或暗槓時不在此限
        \item[(9)] 立直後仍可放棄榮和,但此後視為振聽狀態
        \item[(10)] 立直者在流局時必須宣告聽牌
        \item[(11)] 流局時的供託點棒放置於下一局莊家的右端角落
        \item[(12)] 供託點棒由下一個和了者獲得。若直到半莊結束都無人獲得,則由本局一位取得
        \item[(13)] 立直的詳細步驟規定如下:
        \begin{itemize}
            \item[$\bullet$] 發聲說「立直 (リーチ)」
            \item[$\bullet$] 拿起要切的牌,開始捨牌動作
            \item[$\bullet$] 將牌背完全貼在指定位置,橫向放置
            \item[$\bullet$] 供託千點棒
        \end{itemize}
    \end{itemize}
    \item[39] 副露
    \begin{itemize}
        \item[(1)] 取得他家的捨牌以完成面子的行為稱為副露
        \item[(2)] 副露分為吃、碰、大明槓3種,詳情規定於第40條、第41 條、第42條
        \item[(3)] 同時發聲時的優先順序以碰、大明槓為優先。但若第40條 規定的「吃」發聲正當,而第41、42條規定的「碰、槓」發聲不正當時,以吃優先
        \item[(4)] 亮牌開示方法規定於第43條
        \item[(5)] 對於河底牌不得進行副露
        \item[(6)] 副露時,不得立刻捨棄與亮出的2張牌構成面子的牌 (即禁止食替)
    \end{itemize}
    \item[40] 吃
    \begin{itemize}
        \item[(1)] 取得上家的捨牌以完成順子的行為稱為吃
        \item[(2)] 吃的步驟規定如下
        \begin{itemize}
            \item[$\bullet$] 在第34條第2項規定的相同自摸時機,發聲「吃 (チー)」
            \item[$\bullet$] 公開手牌中的搭子,與剛才上家的捨牌組合完成順子
            \item[$\bullet$] 將完成的順子橫向並列於自己的右端
            \item[$\bullet$] 進行打牌。以此視為吃的成立
        \end{itemize}
        \item[(3)] 打牌前可修正副露牌。但該牌視為暴露牌處理
        \item[(4)] 打牌後若發現指示牌放置錯誤,只有在全員同意或可透過記錄確認時,才允許修正
    \end{itemize}
    \item[41] 碰
    \begin{itemize}
        \item[(1)] 取得他家的捨牌以完成刻子的行為稱為碰
        \item[(2)] 碰的步驟規定如下:
        \begin{itemize}
            \item[$\bullet$] 他家打牌完成後,立刻發聲說「碰 (ポン)」。但對上家的打牌可稍微留有間隔
            \item[$\bullet$] 公開手牌中的對子,與剛才他家的捨牌組合完成刻子
            \item[$\bullet$] 將完成的刻子橫向並列於自己的右端
            \item[$\bullet$] 進行打牌。以此視為碰的成立
        \end{itemize}
        \item[(3)] 打牌前可修正副露牌。但該牌視為暴露牌處理
        \item[(4)] 打牌後若發現指示牌放置錯誤,只有在全員同意或可透過記錄確認時,才允許修正
    \end{itemize}
    \item[42] 槓
    \begin{itemize}
        \item[(1)] 以4張相同牌完成槓子的行為稱為槓
        \item[(2)] 槓的種類分為暗槓、加槓、大明槓3種
        \item[(3)] 所有槓在槓子確定的時間點,寶牌種類皆會增加1個。稱為槓寶牌
        \item[(4)] 槓寶牌指示牌翻開的時間點視為槓的成立
        \item[(5)] 一局中槓的上限為4次
        \item[(6)] 若無法補充王牌 (嶺上牌已無),則無法進行槓牌
